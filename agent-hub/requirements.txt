# agent-hub runtime dependencies
# Install with:  pip install -r requirements.txt
#
# APScheduler is pinned to the 3.x series on purpose: version 4.x is a
# ground-up rewrite with a different API, and all of this code targets 3.x.
APScheduler>=3.10,<4

# Talks to the Telegram Bot API over HTTPS (used by core/notifier.py).
requests>=2.31

# Reads settings.yaml (used by core/config.py).
PyYAML>=6.0

# Loads config/.env into environment variables (used by core/config.py).
python-dotenv>=1.0

# --- news_digest task (services/news/) ---------------------------------------
# Only needed if the news_digest task is enabled in settings.yaml.

# Parses RSS/Atom feeds (used by services/news/fetcher.py).
feedparser>=6.0.12

# The CA bundle used to verify HTTPS feeds. requests already installs certifi
# for its own use, but services/news/fetcher.py imports it directly to hand the
# same bundle to urllib — so it is a real dependency here, not a transitive one.
certifi>=2024.2

# Parses the irregular date strings feeds use when the pre-parsed form is
# missing (used by services/news/fetcher.py).
python-dateutil>=2.9

# Schemas for the digest, and the structured output the LLM must return.
pydantic>=2.13

# Runs the two LLM agents against Ollama's OpenAI-compatible endpoint. The
# "slim" distribution with the openai extra pulls only that one provider
# instead of every model backend pydantic-ai supports.
#
# Pinned to the 2.x series. The original NewsDocsObtainer pinned "<1.0", which
# now resolves to 0.8.1 — that version imports opentelemetry._events, a module
# removed in current opentelemetry-api, so it fails at import time.
pydantic-ai-slim[openai]>=2.0,<3
